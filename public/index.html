// صور → حركة Zoom + انتقالات → دمج مع الصوت
const args = [
  '-i', 'intro.mp4',
  '-i', 'slide-1.jpg',
  '-i', 'voiceover.mp3',
  '-filter_complex', 'zoompan + xfade + concat',
  '-c:v', 'libx264', 'final_output.mp4'
];

await runFfmpeg(args, totalSec, onProgress);
